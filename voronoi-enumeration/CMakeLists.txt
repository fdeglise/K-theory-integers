cmake_minimum_required(VERSION 3.20)
# Some Apple CLT installations omit the usual libc++ header symlink. Locate
# the SDK copy locally rather than changing the user's toolchain installation.
if(CMAKE_HOST_APPLE)
  execute_process(COMMAND xcrun --show-sdk-path OUTPUT_VARIABLE TASK_MAC_SDK OUTPUT_STRIP_TRAILING_WHITESPACE)
  if(EXISTS "${TASK_MAC_SDK}/usr/include/c++/v1")
    set(CMAKE_CXX_FLAGS_INIT "-isystem ${TASK_MAC_SDK}/usr/include/c++/v1")
  endif()
endif()
project(voronoi_cells VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(DEPS_PREFIX "${CMAKE_SOURCE_DIR}/.deps/prefix" CACHE PATH "Local dependency prefix")
list(PREPEND CMAKE_PREFIX_PATH "${DEPS_PREFIX}")
list(PREPEND CMAKE_MODULE_PATH "${CMAKE_SOURCE_DIR}/cmake")
find_path(BOOST_HEADERS boost/multiprecision/gmp.hpp HINTS "${CMAKE_SOURCE_DIR}/.deps/src/boost_1_89_0" REQUIRED)
set(Boost_INCLUDE_DIR "${BOOST_HEADERS}")
find_package(soplex CONFIG REQUIRED)
find_path(SOPLEX_HEADERS soplex/config.h HINTS ${SOPLEX_INCLUDE_DIRS} REQUIRED)
file(STRINGS "${SOPLEX_HEADERS}/soplex/config.h" SOPLEX_EXACT_FLAGS REGEX "^#define SOPLEX_WITH_(BOOST|GMP)$")
list(LENGTH SOPLEX_EXACT_FLAGS SOPLEX_EXACT_FLAG_COUNT)
if(NOT SOPLEX_EXACT_FLAG_COUNT EQUAL 2)
  message(FATAL_ERROR "SoPlex must have GMP and Boost rational support; run scripts/install-deps.sh")
endif()
find_path(GMP_HEADERS gmpxx.h HINTS "${DEPS_PREFIX}/include" REQUIRED)
find_library(GMP_LIBRARY gmp HINTS "${DEPS_PREFIX}/lib" REQUIRED)
find_library(GMPXX_LIBRARY gmpxx HINTS "${DEPS_PREFIX}/lib" REQUIRED)
add_executable(voronoi_cpp cpp/main.cpp cpp/lp_soplex.cpp)
target_include_directories(voronoi_cpp SYSTEM PRIVATE "${BOOST_HEADERS}" "${GMP_HEADERS}")
target_link_libraries(voronoi_cpp PRIVATE libsoplex "${GMPXX_LIBRARY}" "${GMP_LIBRARY}")
target_compile_options(voronoi_cpp PRIVATE -Wall -Wextra)
option(VORONOI_SANITIZE "Enable address and undefined-behavior checks" OFF)
if(VORONOI_SANITIZE)
  target_compile_options(voronoi_cpp PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(voronoi_cpp PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME exact_core COMMAND voronoi_cpp --self-test)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME published_counts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/cpp/check_counts.py" "$<TARGET_FILE:voronoi_cpp>")
set_tests_properties(published_counts PROPERTIES TIMEOUT 1200)
